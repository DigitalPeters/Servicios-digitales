554:async function getEffectiveProductPrice(client, user, product) {
555:  const fallbackPrice = Number(product.price || 0);
556:
557:  if (!user) return fallbackPrice;
558:
559:  if (user.role === "admin") {
560:    return fallbackPrice;
561:  }
562:
563:  // Si este usuario es vendedor de un distribuidor, toma el precio final que su distribuidor definió.
564:  if (user.owner_user_id) {

566:      `SELECT sale_price
567:       FROM subadmin_reseller_prices
568:       WHERE owner_user_id = $1 AND product_id = $2`,
569:      [user.owner_user_id, product.id]

571:
572:    if (resellerPriceResult.rows[0]) {
573:      return Number(resellerPriceResult.rows[0].sale_price || fallbackPrice);
574:    }
575:
576:    // Si no tiene precio final, usa el costo/precio que ese distribuidor tiene contigo.
577:    const ownerPriceResult = await client.query(
578:      `SELECT sale_price
579:       FROM user_product_prices
580:       WHERE user_id = $1 AND product_id = $2`,
581:      [user.owner_user_id, product.id]
582:    );
583:
584:    if (ownerPriceResult.rows[0]) {
585:      return Number(ownerPriceResult.rows[0].sale_price || fallbackPrice);
586:    }
587:
588:    return fallbackPrice;
589:  }
590:
591:  // Si es usuario normal o admin independiente, toma su precio especial si existe.
592:  const customPriceResult = await client.query(
593:    `SELECT sale_price
594:     FROM user_product_prices
595:     WHERE user_id = $1 AND product_id = $2`,
596:    [user.id, product.id]
597:  );

600:    return Number(customPriceResult.rows[0].sale_price || fallbackPrice);
